# Builds FFmpeg's libavutil and libavcodec (configured with only the XMA decoders, see configure.sh)
# without make: list_sources.pl reads FFmpeg's own Makefiles with the configured config.mak and
# this compiles the same files. Used on Windows (scripts/build_ffmpeg_win.bat); iOS uses make.
cmake_minimum_required(VERSION 3.20)
project(ffmpeg_xma C)

set(FFMPEG_SRC "" CACHE PATH "FFmpeg source directory")
set(FFMPEG_CFG "" CACHE PATH "Directory configure.sh ran in (config.h, ffbuild/config.mak)")
if (NOT EXISTS "${FFMPEG_SRC}/libavcodec/avcodec.h" OR NOT EXISTS "${FFMPEG_CFG}/config.h")
    message(FATAL_ERROR "Set FFMPEG_SRC and FFMPEG_CFG")
endif()

find_program(PERL perl REQUIRED)
execute_process(COMMAND "${PERL}" "${CMAKE_CURRENT_SOURCE_DIR}/list_sources.pl" "${FFMPEG_SRC}" "${FFMPEG_CFG}"
    OUTPUT_FILE "${CMAKE_BINARY_DIR}/sources.cmake" RESULT_VARIABLE result)
if (NOT result EQUAL 0)
    message(FATAL_ERROR "list_sources.pl failed")
endif()
include("${CMAKE_BINARY_DIR}/sources.cmake")

# Headers FFmpeg's Makefile generates.
set(GEN "${CMAKE_BINARY_DIR}/gen")
file(STRINGS "${FFMPEG_CFG}/config.h" bigendian REGEX "^#define HAVE_BIGENDIAN ")
string(REGEX REPLACE ".* " "" bigendian "${bigendian}")
file(STRINGS "${FFMPEG_CFG}/config.h" unaligned REGEX "^#define HAVE_FAST_UNALIGNED ")
string(REGEX REPLACE ".* " "" unaligned "${unaligned}")
file(WRITE "${GEN}/libavutil/avconfig.h"
    "/* Generated by scripts/ffmpeg/CMakeLists.txt */\n#ifndef AVUTIL_AVCONFIG_H\n#define AVUTIL_AVCONFIG_H\n"
    "#define AV_HAVE_BIGENDIAN ${bigendian}\n#define AV_HAVE_FAST_UNALIGNED ${unaligned}\n#endif\n")
file(READ "${FFMPEG_SRC}/RELEASE" release)
string(STRIP "${release}" release)
file(WRITE "${GEN}/libavutil/ffversion.h"
    "#ifndef AVUTIL_FFVERSION_H\n#define AVUTIL_FFVERSION_H\n#define FFMPEG_VERSION \"${release}\"\n#endif\n")

foreach (lib avutil avcodec)
    string(TOUPPER ${lib} upper)
    add_library(${lib} STATIC ${FF_${upper}_SOURCES})
    target_include_directories(${lib} PRIVATE "${GEN}" "${FFMPEG_CFG}" "${FFMPEG_SRC}"
        "${FFMPEG_SRC}/compat/atomics/win32" "${FFMPEG_SRC}/compat/stdbit")
    target_compile_definitions(${lib} PRIVATE HAVE_AV_CONFIG_H BUILDING_${lib} _ISOC11_SOURCE
        _FILE_OFFSET_BITS=64 _LARGEFILE_SOURCE WIN32_LEAN_AND_MEAN _USE_MATH_DEFINES
        _CRT_SECURE_NO_WARNINGS _CRT_NONSTDC_NO_WARNINGS _WIN32_WINNT=0x0600)
    # Same flags as configure's MSVC toolchain; /O2 matters: FFmpeg relies on dead-code removal
    # of disabled features ("if (CONFIG_X) ...").
    target_compile_options(${lib} PRIVATE /std:c17 /W0 /O2 /utf-8)
    install(TARGETS ${lib} ARCHIVE DESTINATION lib)
    file(GLOB headers "${FFMPEG_SRC}/lib${lib}/*.h")
    install(FILES ${headers} DESTINATION include/lib${lib})
endforeach()
install(FILES "${GEN}/libavutil/avconfig.h" "${GEN}/libavutil/ffversion.h" DESTINATION include/libavutil)
